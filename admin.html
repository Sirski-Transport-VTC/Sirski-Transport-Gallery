async function updateStatus(id, status) { const action = status === "approved" ? "approve" : "reject"; if ( !confirm( "Are you sure you want to " + action + " this screenshot?" ) ) { return; } try { /* Get the submission first. */ const { data: submission, error: fetchError } = await client .from("gallery_submissions") .select("*") .eq("id", id) .single(); if (fetchError || !submission) { throw new Error( "Unable to find the submission." ); } /* When approving: copy the image from the PRIVATE submissions bucket into the PUBLIC approved bucket. */ if (status === "approved") { const { data: fileData, error: downloadError } = await client.storage .from("gallery-submissions") .download( submission.file_path ); if (downloadError) { throw new Error( "Unable to read the submitted image: " + downloadError.message ); } const originalFilename = submission.file_path .split("/") .pop(); const approvedPath = "approved/" + submission.id + "-" + originalFilename; const { error: uploadError } = await client.storage .from("gallery-approved") .upload( approvedPath, fileData, { cacheControl: "31536000", upsert: false, contentType: fileData.type || "image/jpeg" } ); if (uploadError) { /* If the file already exists, don't create another copy. */ if ( !uploadError.message .toLowerCase() .includes("already exists") && !uploadError.message .toLowerCase() .includes("duplicate") ) { throw new Error( "Unable to publish the image: " + uploadError.message ); } } } /* Now update the submission status. */ const { error: updateError } = await client .from("gallery_submissions") .update({ status: status, reviewed_at: new Date().toISOString() }) .eq("id", id); if (updateError) { throw new Error( "Unable to update submission: " + updateError.message ); } showMessage( status === "approved" ? "Screenshot approved and published to the gallery." : "Submission rejected successfully.", "success" ); await loadSubmissions(); } catch (error) { console.error(error); showMessage( error.message, "error" ); } }
